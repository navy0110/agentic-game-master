def process_request(storage, owner, session, message_id, text, address, provider, audit=None):
    def emit(event, turn, model_call=False):
        if audit:
            try:
                audit(event, turn, model_call)
            except Exception:
                pass  # Diagnostic failure must never prevent a reply or change state.
    text = normalize_message(text, address)
    question = text.lower()
    if question.strip('!?. ') in {'hi', 'hello', 'hey', 'hola', 'help',
            'what can you do', 'how does this work', 'how can you help me'}:
        emit('welcome', 0)
        return ('I am Agentic Game Master, a preparation assistant for tabletop GMs and '
                'RPG developers. Give me your world rules and ask for a quest, an NPC '
                'or a dynamic event. Then send player choices to continue the scene. '
                'I generate text and save campaign facts in this conversation; I do '
                'not control a live game. Try: "Create a harbor official with motives, '
                'a secret and three lines of dialogue." Use /help for commands. '
                'In ASI:One, select @agenticgamemaster as the recipient for every message.')
    if (not question.startswith('/') and '/gm-status' in question
            and any(term in question for term in ('which command', 'what extra',
                                                   'what does', 'difference between'))):
        emit('command-guide', 0)
        return ('Use /status while players can see your screen. It shows only the saved '
                'turn, fact count and open-thread count; it hides all story details. '
                'It is not a narrative recap. Read /gm-status privately: it shows the '
                'full recorded facts and open threads/proposals, including NPC secrets '
                'and unrevealed hooks. Both commands are read-only: neither advances '
                'the story nor calls the model. These are display modes, not separate '
                'GM/player access permissions.')
    key = 'agm:v1:' + hashlib.sha256(json.dumps([owner, session]).encode()).hexdigest()
    saved = storage.get(key)
    try:
        record = json.loads(saved) if isinstance(saved, str) else saved
    except json.JSONDecodeError as exc:
        raise ValueError('storage-json: saved record is not valid JSON') from exc
    record = record or {'state': None, 'replies': {}}
    state = record['state']
    turn = state['turn'] if state else 0
    if not text:
        return 'Describe your world, rules, characters, and desired adventure to begin.'
    if text.lower() == '/help':
        emit('help', turn)
        return ('Agentic Game Master\nDescribe a world to begin; send player decisions to continue.\n'
                '/status: spoiler-safe campaign summary, without advancing the story\n'
                '/gm-status: full GM record, INCLUDING SECRETS\n'
                '/export: full campaign JSON, INCLUDING SECRETS\n'
                '/review [notes]: ask the separate Lorekeeper agent for a read-only GM consistency review; may contain secrets\n'
                '/help: show commands\n'
                'Explicitly identify incorrect facts when requesting a correction. '
                'A new conversation/session starts a separate campaign.')
    if text.lower() == '/status':
        emit('status', turn)
        return render_player_status(state)
    if text.lower() == '/gm-status':
        emit('gm-status', turn)
        return render_status(state)
    if text.lower() == '/export':
        emit('export', turn)
        return json.dumps(state, ensure_ascii=False, indent=2) if state else 'No campaign yet.'
    if text.startswith('/'):
        emit('unknown-command', turn)
        return 'Unknown command. Send /help. Your campaign is unchanged.'
    if len(text) > 16000:
        raise ValueError('Request too long. Send at most 16000 characters.')
    fingerprint = hashlib.sha256(text.encode()).hexdigest()
    cached = record['replies'].get(str(message_id))
    if cached:
        if cached['fingerprint'] != fingerprint:
            raise ValueError('Message ID conflict.')
        emit('duplicate-replay', turn)
        return cached['text']
    emit('generation-start', turn, True)
    result = advance_state(state, text, provider)
    reply = result['adventure']
    if result['changes']:
        reply += '\n\nConsequences:\n' + '\n'.join('- ' + change for change in result['changes'])
    record['state'] = result
    record['replies'][str(message_id)] = {'fingerprint': fingerprint, 'text': reply}
    encoded = json.dumps(record, ensure_ascii=False)
    if len(encoded.encode()) > 200000:
        raise ValueError('Campaign storage limit reached. Export before starting another chapter.')
    # A single storage write keeps the campaign and duplicate response together.
    storage.set(key, encoded)
    emit('generation-saved', result['turn'], True)
    return reply


# The execution environment injects this flag only in the local test harness.
if not globals().get('_AGM_TEST_MODE', False):
    import os
    import requests
    import asyncio
    import weakref
    import time
    from datetime import datetime, timezone
    from uuid import uuid4
    from uagents import Agent, Context, Protocol
    from uagents_core.contrib.protocols.chat import (
        ChatMessage, ChatAcknowledgement, TextContent, chat_protocol_spec,
    )

    # Hosted Agent() wraps the platform-provided identity; never pass a seed here.
    agent = Agent()
    protocol = Protocol(spec=chat_protocol_spec)
    review_protocol = Protocol(name='AGMLoreReview', version='1.0.0')
    # Keep network I/O off the event loop, but serialize each campaign's writes.
    campaign_locks = weakref.WeakValueDictionary()

    class ModelUnavailable(RuntimeError):
        pass

    class ModelTimeout(RuntimeError):
        pass

    def hosted_model(messages, audit=None):
        key = os.environ.get('ASI1_API_KEY') or os.environ.get('ASI_ONE_API_KEY')
        base = os.environ.get('ASI1_BASE_URL', 'https://api.asi1.ai/v1').rstrip('/')
        if not key or not base.startswith('https://'):
            raise RuntimeError('Hosted ASI credentials are unavailable.')
        messages, response_format, npc_mode = generation_options(messages)
        # One retry only. Network inactivity limits are not total wall-clock limits.
        # A timed-out inference may still use provider quota; no state is saved here.
        for attempt, read_timeout in enumerate((40, 20)):
            started = time.monotonic()
            def report(outcome):
                if audit:
                    try:
                        audit(f'provider-{outcome}-attempt-{attempt + 1}-elapsed-{time.monotonic() - started:.1f}s', 0, True)
                    except Exception:
                        pass
            report('start')
            try:
                response = requests.post(base + '/chat/completions',
                    headers={'Authorization': 'Bearer ' + key, 'Content-Type': 'application/json'},
                    json={'model': 'asi1', 'messages': messages, 'max_tokens': 3000,
                          'temperature': 0.7, 'response_format': response_format,
                          'reasoning_effort': 'low'},
                    timeout=(5, read_timeout))
                response.raise_for_status()
                report('received')
                break
            except requests.Timeout as exc:
                report('timeout')
                if attempt:
                    raise ModelTimeout('Model response timed out after one retry') from exc
            except requests.RequestException as exc:
                report('http-error')
                status = getattr(getattr(exc, 'response', None), 'status_code', None)
                transient = isinstance(exc, requests.ConnectionError) or status in {502, 503, 504}
                if attempt or not transient:
                    raise ModelUnavailable(f'Model unavailable; HTTP status={status}') from exc
            time.sleep(0.5)
        try:
            result = response.json()
        except ValueError as exc:
            raise ValueError('provider-envelope: HTTP body is not JSON') from exc
        return completion_text(result, npc_mode)

    async def request_lore_review(ctx, sender, session, message_id, extra):
        peer = os.environ.get('LOREKEEPER_ADDRESS') or LOREKEEPER_ADDRESS
        if not peer or peer == agent.address:
            return ('Lorekeeper is not configured. Your campaign is unchanged. '
                    'The operator must deploy the separate reviewer and configure its address.')
        snapshot = review_snapshot(ctx.storage, sender, session, extra)
        if snapshot is None:
            return ('Create an NPC or scene first, then send /review in the same conversation. '
                    'Or send /review followed by a short rules-and-draft excerpt. '
                    'This sends the material to Lorekeeper and may include GM secrets.')
        digest = hashlib.sha256(snapshot.encode()).hexdigest()
        request_id = str(message_id)
        reference = hashlib.sha256(request_id.encode()).hexdigest()[:12]
        ctx.logger.info(f'AGM event=review-sent; reference={reference}; peer={peer}; snapshot={digest[:12]}')
        try:
            response, status = await asyncio.wait_for(ctx.send_and_receive(
                peer, LoreReviewRequest(request_id=request_id, snapshot_json=snapshot),
                response_type=LoreReviewResponse, timeout=22), timeout=25)
            ctx.logger.info(f'AGM event=review-delivery; reference={reference}; response_type={type(response).__name__}; delivery={getattr(status, "status", "unknown")}')
            if (not isinstance(response, LoreReviewResponse) or response.error
                    or response.request_id != request_id or response.snapshot_hash != digest):
                raise ValueError('Review response did not match the request.')
            report = parse_review(response.report_json, snapshot)
        except Exception as exc:
            ctx.logger.info(f'AGM event=review-unavailable; reference={reference}; type={type(exc).__name__}')
            return ('Lorekeeper could not complete a validated review. '
                    'Your existing content and campaign are unchanged; no review is claimed. '
                    'You can continue playing or retry /review later. Reference: ' + reference)
        ctx.logger.info(f'AGM event=review-received; reference={reference}; peer={peer}; snapshot={digest[:12]}')
        return render_review(report, peer, reference)

    @protocol.on_message(ChatMessage)
    async def handle_message(ctx: Context, sender: str, msg: ChatMessage):
        await ctx.send(sender, ChatAcknowledgement(timestamp=datetime.now(timezone.utc),
                                                  acknowledged_msg_id=msg.msg_id))
        text = '\n'.join(item.text for item in msg.content if isinstance(item, TextContent))
        message_tag = hashlib.sha256(str(msg.msg_id).encode()).hexdigest()[:12]
        def audit(event, turn, model_call):
            ctx.logger.info(f'AGM event={event}; message={message_tag}; turn={turn}; model_call={str(model_call).lower()}')
        def provider(messages):
            return hosted_model(messages, audit=audit)
        try:
            lock_key = (sender, str(ctx.session))
            lock = campaign_locks.get(lock_key)
            if lock is None:
                lock = asyncio.Lock()
                campaign_locks[lock_key] = lock
            async with lock:
                normalized = normalize_message(text, agent.address)
                if normalized.split(maxsplit=1)[0:1] == ['/review']:
                    extra = normalized[len('/review'):].strip()
                    reply = await request_lore_review(ctx, sender, str(ctx.session), str(msg.msg_id), extra)
                else:
                    reply = await asyncio.to_thread(process_request, ctx.storage, sender,
                                        str(ctx.session), str(msg.msg_id), text,
                                        agent.address, provider, audit)
        except Exception as exc:
            # Never log provider payloads, API keys, or user campaign data.
            ctx.logger.error(f'Campaign failed ({type(exc).__name__}); check hosted logs/quotas.')
            if isinstance(exc, ModelTimeout):
                reply = ('The story model took too long to respond, including one automatic '
                         'retry. Please try again later. You do not need to create a campaign '
                         'first. /help and /status remain available without the model. '
                         'No successful campaign update was confirmed.')
            elif isinstance(exc, ModelUnavailable):
                ctx.logger.error(f'AGM provider failure: {exc}')
                reply = ('The story model is temporarily unavailable or its account quota '
                         'has been reached. I could not generate your content. Please try '
                         'again later; the operator should check model access and quota. '
                         '/help and /status do not need the model.')
            elif isinstance(exc, ValueError):
                # Validation messages are internal constants/positions, not model text.
                ctx.logger.error(f'AGM validation failure: {exc}')
                reply = ('The generated response could not be validated or saved. '
                         'Please retry once; if it repeats, share this reference with '
                         'the operator: ' + message_tag + '. No successful campaign '
                         'update was confirmed. Your request does not require prior setup.')
            else:
                reply = ('I encountered a technical error before completing your request. '
                         'No successful campaign update was confirmed. Please retry; '
                         'if it repeats, share this reference with the operator: ' + message_tag)
        await ctx.send(sender, ChatMessage(timestamp=datetime.now(timezone.utc), msg_id=uuid4(),
                                          content=[TextContent(type='text', text=reply)]))

    @protocol.on_message(ChatAcknowledgement)
    async def handle_ack(ctx: Context, sender: str, msg: ChatAcknowledgement):
        pass

    @review_protocol.on_message(LoreReviewResponse)
    async def handle_unsolicited_review(ctx: Context, sender: str, msg: LoreReviewResponse):
        # Solicited responses are matched by send_and_receive to peer and session.
        # Late/unsolicited results must not become story turns or alter storage.
        ctx.logger.info('AGM event=unsolicited-review-ignored')

    agent.include(protocol, publish_manifest=True)
    agent.include(review_protocol, publish_manifest=True)
